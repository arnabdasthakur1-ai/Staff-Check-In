const SPREADSHEET_ID = '19nsXvIcQbtZj3nTDxXsWKT8AMtizYmYGm5peK3NkRDc'; const SHEET_NAME = 'CheckIns'; const DRIVE_FOLDER_ID = '1MwA5kl8Ei8M7Ld3C8Q9bdKocM53kbutX'; function doPost(e) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const errSheet = ss.getSheetByName('Errors') || ss.insertSheet('Errors'); try { const data = JSON.parse(e.postData.contents); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(['Timestamp', 'Name', 'Hub', 'Latitude', 'Longitude', 'Accuracy', 'Photo URL']); } const base64 = data.photo.split(',')[1]; const blob = Utilities.newBlob(Utilities.base64Decode(base64), 'image/jpeg', data.name.replace(/\s+/g, '_') + '_' + new Date().getTime() + '.jpg'); const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); sheet.appendRow([ data.timestamp, data.name, data.hub, data.lat, data.lng, data.accuracy, file.getUrl() ]); return ContentService.createTextOutput(JSON.stringify({ result: 'success' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { errSheet.appendRow([new Date().toISOString(), err.toString(), e.postData ? e.postData.contents : 'NO POSTDATA']); return ContentService.createTextOutput(JSON.stringify({ result: 'error', message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
